// Доступные переменные контекста:
// Данные					- произвольный	- данные пакета данных.
// ПараметрыПакетаДанных	- структура	- структура, содержащая информацию о пакете данных.
// ПараметрыСпособаОбработки- структура	- структура, содержащая параметры способа обработки на момент вызова процедуры обработки пакета.
 
НачатьТранзакцию();    
КоличествоОбъектовобработано = 0;  
// Предполагается, что Данные - это коллекция, содержащая элементы со свойством Ссылка.
Для каждого ЭлементДанных Из Данные Цикл 
	Объект = ЭлементДанных.Ссылка.ПолучитьОбъект();	
	Если Объект = Неопределено Тогда
		Продолжить;
	КонецЕсли;   
	 
	Объект.Заблокировать();                        
	Объект.ОбменДанными.Загрузка = Истина;	
	Объект.Удалить();		
	КоличествоОбъектовобработано = КоличествоОбъектовобработано + 1;	
КонецЦикла;
ЗафиксироватьТранзакцию();

// РезультатОбработкиПакета - структура с описанием статуса завершения обработки. Содержит свойства:
//	СостояниеПакетаДанных:
//			Перечисления.ум_СостоянияПакетаДанных.Обработан - пакет успешно обработан
//			Перечисления.ум_СостоянияПакетаДанных.Ошибка	- ошибка обработки пакета
//			Перечисления.ум_СостоянияПакетаДанных.КОбработке- пакет в данной попытке успешно обработан не был, необходима повторная обработка пакета.
//	Сообщение			- Строка -	произвольнный текст, сохраняемый в журале событий пакетов данных. Значение не обязательное.
//	РезультатОбработки	- произвольный - набор данных, сохраняемый в журнале событий пакетов данных. Значение не обязательное. 
РезультатОбработкиПакета = Новый Структура;
РезультатОбработкиПакета.Вставить("СостояниеПакетаДанных"	, Перечисления.ум_СостоянияПакетаДанных.Обработан);
РезультатОбработкиПакета.Вставить("Сообщение"				, "Объектов обработано: "+КоличествоОбъектовобработано);
РезультатОбработкиПакета.Вставить("РезультатОбработки"		, Неопределено);
